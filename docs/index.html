💳 1. Snowflake Medallion Pipeline

Challenges & Lessons Learned:
Handling out-of-order raw JSON logs without creating duplicate Silver records was solved using Snowflake Streams + QUALIFY ROW_NUMBER() OVER (PARTITION BY transaction_id ORDER BY tx_timestamp DESC) = 1.

CREATE OR REPLACE TASK silver_tx_task SCHEDULE = '1 MINUTE' WHEN SYSTEM$STREAM_HAS_DATA('bronze_tx_stream') AS INSERT INTO silver_clean_transactions SELECT transaction_id, customer_id, card_number, amount, tx_timestamp FROM bronze_tx_stream QUALIFY ROW_NUMBER() OVER (PARTITION BY transaction_id ORDER BY tx_timestamp DESC) = 1;
CREATE OR REPLACE MASKING POLICY mask_card_number AS (val string) -> CASE WHEN CURRENT_ROLE() IN ('COMPLIANCE_ADMIN') THEN val ELSE CONCAT('XXXX-XXXX-XXXX-', RIGHT(val, 4)) END;
